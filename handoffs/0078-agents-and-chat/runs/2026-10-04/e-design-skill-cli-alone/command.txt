# The design skill with the CLI alone, 2026-10-04. It was meant to have the plugin's MCP server too, but the command
# passed --strict-mcp-config, which keeps a plugin's server from loading ("MCP servers: none" in the transcript). So this
# run is the other half of the criterion: the skill still does every step with the grooph command and no tools.
# A small project in a temporary folder (<scratch>/chat-e: git init, a package.json, one source file); the plugin
# loaded from this branch; the grooph command on PATH pointing at this branch's build. Model claude-sonnet-5-5.
PATH=<scratch>/bin-e:/opt/homebrew/bin:/usr/bin:/bin claude -p "$(cat prompt.txt)" --model claude-sonnet-5-5 \
  --plugin-dir <repo>/plugins/grooph --setting-sources project,local --strict-mcp-config \
  --allowedTools "mcp__plugin_grooph_grooph" "Bash(grooph:*)" Read Write Edit Skill Glob Grep \
  --max-budget-usd 3 --output-format stream-json --verbose > local/stream.jsonl
