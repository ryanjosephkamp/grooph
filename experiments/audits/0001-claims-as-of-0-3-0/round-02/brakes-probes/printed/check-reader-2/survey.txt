
== patterns/debate-then-build.grooph.json
  checks: tests  critics: judge  stops: done:success  gates: plan-gate
  loop debate[planner-a,planner-b,judge] back=e-judge-rebut mode=judgment bar=y stops=bar-passed|max-iterations>judge|budget
  loop build[builder,tests] back=e-tests-fail mode=grind bar=n stops=max-iterations|budget
  e-builder-tests: builder -> tests 
  e-tests-fail: tests -> builder fail ev=["test output"]
  e-tests-pass: tests -> done pass
